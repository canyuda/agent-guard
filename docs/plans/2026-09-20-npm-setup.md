**Goal:** 把 agent-guard 发布为 npm 全局 CLI(`<pkg名>`),提供 `agent-guard setup` 四步向导(key → 工具多选 → 作用域 → 确认)自动挂载 ZCode/Claude Code/Cursor 的 PreToolUse hook。

**Tech Stack:** Node ≥20 ESM,零 npm 依赖,`node:test`;交互用 `node:readline/promises`。

```js
import { os } from "node:os"; // 实际写:import { homedir } from "node:os"
import { join, dirname } from "node:path";
import { existsSync, copyFileSync, readFileSync, mkdirSync } from "node:fs";
import { fileURLToPath } from "node:url";

export const PKG_ROOT = join(dirname(fileURLToPath(import.meta.url)), "..");

export function resolvePaths(opts = {}) {
  const home = opts.home ?? homedir();
  const env = opts.env ?? process.env;
  const pkgRoot = opts.pkgRoot ?? PKG_ROOT;
  const dataDir = join(home, ".agent-guard");
  return {
    dataDir,
    configPath: env.AGENT_GUARD_CONFIG ?? join(dataDir, "config.json"),
    cachePath: join(dataDir, "cache", "judgments.json"),
    logPath: join(dataDir, "logs", "audit.jsonl"),
  };
}

export function ensureUserConfig(paths, pkgRoot = PKG_ROOT) {
  if (paths.configPath !== join(paths.dataDir, "config.json")) return; // env 指定或非默认,不代管
  try {
    JSON.parse(readFileSync(paths.configPath, "utf8"));
    return; // 已有合法用户配置,不覆盖
  } catch { /* 无/损坏 → 尝试从包内复制 */ }
  try {
    mkdirSync(paths.dataDir, { recursive: true });
    copyFileSync(join(pkgRoot, "config.json"), paths.configPath);
  } catch { /* 包内无默认则留空,loadConfig 自有兜底 */ }
}
```

`lib/config.mjs`:`loadConfig` 默认参数改 `resolvePaths().configPath` 并调用 `ensureUserConfig`;`lib/judge.mjs` 默认 `cachePath = resolvePaths().cachePath`;`lib/log.mjs` 默认 `p = isAbsolute ? : resolvePaths().logPath`。
